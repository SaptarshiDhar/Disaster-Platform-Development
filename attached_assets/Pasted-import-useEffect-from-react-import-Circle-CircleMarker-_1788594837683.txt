import { useEffect } from "react";

import {
  Circle,
  CircleMarker,
  MapContainer,
  Polygon,
  TileLayer,
  Tooltip,
  useMap,
} from "react-leaflet";

import {
  hazardPolygons,
  RISK_COLORS,
} from "../../data/mockHazards";

import {
  candidateSites,
  incidentReports,
} from "../../data/mockDashboard";

function MapController({ region }) {
  const map = useMap();

  useEffect(() => {
    map.flyTo(
      region.center,
      region.zoom,
      {
        duration: 1.2,
      }
    );
  }, [map, region]);

  return null;
}

function DisasterMap({
  activeHazard,
  habitations,
  selectedHabitation,
  onSelectHabitation,
  layers,
  region,
}) {
  const polygons =
    activeHazard === "exposure"
      ? []
      : hazardPolygons[activeHazard] || [];

  const getHabitationColor = (
    habitation
  ) => {
    if (activeHazard === "exposure") {
      return "#a855f7";
    }

    const hazard =
      habitation.hazards[activeHazard];

    return (
      RISK_COLORS[hazard?.level] ||
      "#38bdf8"
    );
  };

  return (
    <div className="map-container-wrapper">
      <MapContainer
        center={[22.9734, 78.6569]}
        zoom={5}
        minZoom={4}
        maxZoom={14}
        className="disaster-map"
      >
        <MapController region={region} />

        <TileLayer
          attribution='&copy; OpenStreetMap contributors'
          url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
        />

        {/* MOCK HAZARD POLYGONS */}

        {polygons.map((zone) => (
          <Polygon
            key={zone.id}
            positions={zone.positions}
            pathOptions={{
              color:
                RISK_COLORS[zone.risk],
              fillColor:
                RISK_COLORS[zone.risk],
              fillOpacity: 0.35,
              weight: 1.4,
            }}
          >
            <Tooltip sticky>
              <strong>{zone.name}</strong>
              <br />
              Prototype Risk: {zone.risk}
            </Tooltip>
          </Polygon>
        ))}

        {/* MOCK POPULATION OVERLAY */}

        {layers.population &&
          habitations.map(
            (habitation) => (
              <Circle
                key={`population-${habitation.id}`}
                center={
                  habitation.coordinates
                }
                radius={
                  12000 +
                  habitation.population * 4
                }
                pathOptions={{
                  color: "#a855f7",
                  fillColor: "#a855f7",
                  fillOpacity: 0.12,
                  opacity: 0.2,
                  weight: 1,
                }}
              />
            )
          )}

        {/* HABITATIONS */}

        {layers.habitations &&
          habitations.map(
            (habitation) => {
              const selected =
                selectedHabitation?.id ===
                habitation.id;

              return (
                <CircleMarker
                  key={habitation.id}
                  center={
                    habitation.coordinates
                  }
                  radius={
                    selected ? 10 : 7
                  }
                  pathOptions={{
                    color: selected
                      ? "#ffffff"
                      : getHabitationColor(
                          habitation
                        ),

                    fillColor:
                      getHabitationColor(
                        habitation
                      ),

                    fillOpacity: 0.95,

                    weight: selected
                      ? 3
                      : 1.5,
                  }}
                  eventHandlers={{
                    click: () =>
                      onSelectHabitation(
                        habitation
                      ),
                  }}
                >
                  <Tooltip>
                    <strong>
                      {habitation.name}
                    </strong>

                    <br />

                    {habitation.district},{" "}
                    {habitation.state}

                    <br />

                    Population:{" "}
                    {habitation.population.toLocaleString()}
                  </Tooltip>
                </CircleMarker>
              );
            }
          )}

        {/* INCIDENT REPORTS */}

        {layers.incidents &&
          incidentReports.map(
            (incident) => (
              <CircleMarker
                key={incident.id}
                center={
                  incident.coordinates
                }
                radius={5}
                pathOptions={{
                  color: "#facc15",
                  fillColor: "#ef4444",
                  fillOpacity: 1,
                  weight: 2,
                }}
              >
                <Tooltip>
                  <strong>
                    {incident.type}
                  </strong>

                  <br />

                  {incident.place}

                  <br />

                  Reported: {incident.time}
                </Tooltip>
              </CircleMarker>
            )
          )}

        {/* CANDIDATE SITES */}

        {layers.candidateSites &&
          candidateSites.map((site) => (
            <CircleMarker
              key={site.id}
              center={site.coordinates}
              radius={8}
              pathOptions={{
                color: "#86efac",
                fillColor: "#22c55e",
                fillOpacity: 0.9,
                weight: 2,
              }}
            >
              <Tooltip>
                <strong>{site.name}</strong>

                <br />

                Candidate site — prototype only.
              </Tooltip>
            </CircleMarker>
          ))}
      </MapContainer>

      <div className="map-risk-legend">
        {activeHazard === "exposure" ? (
          <>
            <strong>
              Population Exposure
            </strong>

            <span>
              <i className="legend-dot population" />
              Prototype population overlay
            </span>

            <small>
              Future: WorldPop-derived layer
            </small>
          </>
        ) : (
          <>
            <strong>Risk Level</strong>

            <span>
              <i className="legend-dot very-high" />
              Very High
            </span>

            <span>
              <i className="legend-dot high" />
              High
            </span>

            <span>
              <i className="legend-dot moderate" />
              Moderate
            </span>

            <span>
              <i className="legend-dot low" />
              Low
            </span>
          </>
        )}
      </div>
    </div>
  );
}

export default DisasterMap;
